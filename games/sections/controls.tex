\section{Controles}

\begin{itemize}
    \item WASD - Faz o jogador se movimentar
    \item \textit{Left Mouse Button} (LMB) [no mundo]
    \begin{itemize}
        \item Ataca quando a espada está equipada na mão 
        do jogador
        \item Marca inimigo a ser atacado quando espada 
        flutua perto do jogador
    \end{itemize}
    \item \textit{Left Mouse Button} (LMB) [na interface] -
    utiliza elementos da interface, seleciona nós da 
    \textit{skill tree}, interage com as espadas na mochila
    e durante o combate.
    \item \textit{Right Mouse Button} (RMB) - mostra 
    informações sobre elementos do jogo
\end{itemize}
