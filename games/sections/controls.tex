\section{Controles}

\begin{itemize}
    \item WASD 
    \begin{itemize}
        \item Movimentar o jogador
    \end{itemize}

    \item \textit{Left Mouse Button} (LMB) [no mundo]
    \begin{itemize}
        \item Atacar quando a espada estiver equipada na 
        mão do jogador
        \item Marcar o inimigo a ser atacado quando a 
        espada estiver flutuando próxima ao jogador
    \end{itemize}
    
    \item \textit{Left Mouse Button} (LMB) [na interface]
    \begin{itemize}
        \item Interagir com elementos da interface
        \item Selecionar nós da \textit{skill tree}
        \item Interagir com as espadas na mochila
        \item Interagir com as espadas durante o combate
        \item Interagir com a caixa de diálogo
    \end{itemize}

    \item \textit{Right Mouse Button} (RMB)
    \begin{itemize}
        \item Mostrar informações sobre elementos do jogo
    \end{itemize}
    
    \item E
    \begin{itemize}
        \item Avançar o texto da caixa de diálogo
        \item Confirmar a seleção na caixa de diálogo
        \item Confirmar a interação com elementos do mapa
    \end{itemize}

    \item Esc
    \begin{itemize}
        \item Abrir e fechar o menu de pausa durante o jogo
    \end{itemize}

    \item Q
    \begin{itemize}
        \item Abrir e fechar a mochila
    \end{itemize}
\end{itemize}
